How to Use in React

    
me-react.js contains the full editor engine — this single file is all you need.


    

      
Direct CDN include Recommended

      
<script src="https://unpkg.com/react@18/umd/react.production.min.js"></script>
<script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>
<script src="me-react.js"></script>

<script>
  const e = React.createElement;
  function App() {
    const [md, setMd] = React.useState('# Hello\n\nThis is **bold**.');
    return e(MEReact.MeEditor, {
      value: md,               // controlled Markdown value
      onChange: setMd,         // fired on every edit
      options: { title: 'My Editor' }
    });
  }
  ReactDOM.createRoot(document.getElementById('root')).render(e(App));
</script>

    


    

      
Props

      
{
  value: '# md',              // controlled Markdown value
  onChange: fn,               // called with Markdown on every edit
  options: {
    title: 'ME React Editor', // nav bar title
    uploadUrl: '/upload',     // paste-image upload endpoint
    showViewCode: true,       // toolbar View-Markdown button
    showCopy: true,           // toolbar Copy button
    showAudio: true,          // toolbar Audio button
    showVideo: true,          // toolbar Video button
    showStatusBar: true,      // character count bar
    maxLength: 0              // cap; count turns red when exceeded
  }
}
