How to Use in Vue

me-vue.js contains the full editor engine — this single file is all you need.

Direct CDN include Recommended

<script src="https://unpkg.com/vue@3"></script>
<script src="me-vue.js"></script>

<div id="app">
  <!-- v-model two-way binding: edits sync automatically -->
  <me-editor v-model="content" title="My Editor" />
</div>
<script>
  Vue.createApp({
    data: () => ({ content: '# Hello\n\nThis is **bold**.' })
  }).use(MEVue).mount('#app');
</script>

Props & events

<me-editor
  v-model="content"          // Markdown, two-way bound
  :options="{
    title: 'ME Vue Editor',  // nav bar title
    uploadUrl: '/upload',    // paste-image upload endpoint
    showViewCode: true,      // toolbar View-Markdown button
    showCopy: true,          // toolbar Copy button
    showAudio: true,         // toolbar Audio button
    showVideo: true,         // toolbar Video button
    showStatusBar: true,     // character count bar
    maxLength: 0             // cap; count turns red when exceeded
  }"
  @change="on_change"        // fired on every edit with Markdown
/>

Programmatic API

// via a template ref
this.$refs.editor.setMarkdown('# New content');
this.$refs.editor.getMarkdown();
this.$refs.editor.importFromElement('#source-div');
this.$refs.editor.download('doc', 'md');  // 'md' | 'html' | 'mixed'

// core modules are also exported for direct use
MEVue.MdParser.parse('# md -> html');
MEVue.MdSerializer.serialize(domNode);
